<script>
function openFlyerModal(src, captionText) {
  var modal = document.getElementById('flyerModal');
  var img = document.getElementById('modalFlyerImg');
  var caption = document.getElementById('modalFlyerCaption');
  img.src = src;
  caption.textContent = captionText;
  modal.classList.add('active');
  document.body.style.overflow = 'hidden';
}

function closeFlyerModal(event) {
  var modal = document.getElementById('flyerModal');
  if (event.target === modal || event.target.classList.contains('insta-modal-close')) {
    modal.classList.remove('active');
    document.body.style.overflow = '';
  }
}

function toggleLike(btn) {
  var countSpan = btn.querySelector('.like-count');
  var current = parseInt(countSpan.textContent, 10);
  if (btn.classList.contains('liked')) {
    btn.classList.remove('liked');
    countSpan.textContent = current - 1;
  } else {
    btn.classList.add('liked');
    countSpan.textContent = current + 1;
  }
}
</script>
